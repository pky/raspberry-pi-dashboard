```bash
# /boot/firmware/cmdline.txt (カーネルパラメーター)

pcie_aspm=off pcie_port_pm=off
pci=pcie_bus_perf

# /boot/firmware/config.txt (PCIe設定)
dtparam=pciex1_gen=3
dtparam=pciex1
dtparam=pciex1_no_l0s=1

# /etc/fstab (ファイルシステム最適化)
```
