# M.2 SSD SMART監視システム - Cronジョブ設定

## 週1回のSSD健康テスト（毎週日曜日 午前2時）
0 2 * * 0 /path/to/raspberry-pi-dashboard/scripts/ssd_management/ssd_weekly_test.sh

## データ同期（毎日午前3時）
0 3 * * * /path/to/raspberry-pi-dashboard/scripts/ssd_management/sync_cron.sh

# 設定方法:
# 1. crontabエディタを開く
#    crontab -e

# 2. 上記の行を追加して保存
#
# 3. 設定確認
#    crontab -l
#
# 4. cronサービス再起動（必要に応じて）
#    sudo systemctl restart cron
